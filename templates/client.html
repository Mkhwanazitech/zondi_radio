function initMap(){map=L.map('map').setView([-25.7,28.1],10); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map); refreshLocs();} async function refreshLocs(){let r=await fetch('/get_locations'); let locs=await r.json(); allLocs=locs; let h=''; for(let u in locs){let l=locs[u]; h+=`
${u}: ${l.lat.toFixed(4)},${l.lng.toFixed(4)} at ${l.time} Maps
`; if(map){if(!markers[u]) markers[u]=L.marker([l.lat,l.lng]).addTo(map).bindPopup(u); else markers[u].setLatLng([l.lat,l.lng]);}} document.getElementById('clientList').innerHTML=h||'No clients yet';} async function loadGroups(){let r=await fetch('/api/groups'); groups=await r.json(); let h=''; for(let id in groups){let g=groups[id]; h+=`
${g.name}${g.messages.length} msgs
`;} document.getElementById('groupList').innerHTML=h;} async function openChat(id){cur=id; socket.emit('join_group',{group:id}); let r=await fetch('/api/group_messages/'+id); groups[id].messages=await r.json(); renderChat();} function renderChat(){if(!cur) return; let g=groups[cur]; let area=document.getElementById('chatArea'); area.innerHTML=''; g.messages.forEach(m=>{let me=m.user==='{{user}}'; let d=document.createElement('div'); d.className='bub '+(me?'me':'other'); if(m.type=='audio'){d.innerHTML=`
${m.user}
${m.time}
`;} else {d.innerHTML=`
${m.user}
${m.text}
${m.time} ✓✓
`;} area.appendChild(d);}); area.scrollTop=area.scrollHeight;} async function sendText(){let t=document.getElementById('msgInput').value.trim(); if(!t||!cur) return; document.getElementById('msgInput').value=''; await fetch('/api/send_text',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({group:cur,text:t})});} async function toggleVoice(){let btn=document.getElementById('micBtn'); if(!isRec){try{let s=await navigator.mediaDevices.getUserMedia({audio:true}); chunks=[]; rec=new MediaRecorder(s,{mimeType:'audio/webm'}); rec.ondataavailable=e=>{if(e.data.size>0) chunks.push(e.data)}; rec.onstop=async()=>{let blob=new Blob(chunks,{type:'audio/webm'}); if(blob.size<800) return; let fd=new FormData(); fd.append('voice',blob,'v.webm'); fd.append('group',cur); await fetch('/api/send_voice',{method:'POST',body:fd});}; rec.start(); isRec=true; btn.innerText='◼';}catch(e){alert(e.message)}} else {rec.stop(); rec.stream.getTracks().forEach(t=>t.stop()); isRec=false; btn.innerText='🎙️';}} async function loadPanics(){let r=await fetch('/get_panic_alerts'); let alerts=await r.json(); let h=''; alerts.reverse().forEach(a=>{h+=`

`;}); document.getElementById('panicList').innerHTML=h||'No SOS yet';} socket.on('new_msg', d=>{if(groups[d.group]){groups[d.group].messages.push(d.msg); if(cur==d.group) renderChat(); loadGroups();}}); socket.on('location_update', d=>{allLocs[d.user]=d.location; if(map){let l=d.location; if(!markers[d.user]) markers[d.user]=L.marker([l.lat,l.lng]).addTo(map).bindPopup(d.user); else markers[d.user].setLatLng([l.lat,l.lng]);} refreshLocs();}); socket.on('panic_alert', d=>{loadPanics(); alert('🚨 SOS FROM '+d.user); if(map){map.setView([d.location.lat,d.location.lng],15);}}); setInterval(refreshLocs,5000); loadGroups(); loadPanics();
